cmake_minimum_required(VERSION 3.25)
project(MythCoolMini VERSION 0.2.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SNAPPY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SNAPPY_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(SNAPPY_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/snappy EXCLUDE_FROM_ALL)
set(HAVE_SNAPPY ON CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/leveldb EXCLUDE_FROM_ALL)
target_include_directories(leveldb PRIVATE third_party/snappy "${CMAKE_BINARY_DIR}/third_party/snappy")
add_library(PawnIOLib SHARED third_party/PawnIO/PawnIOLib/PawnIOLib.cpp)
target_include_directories(PawnIOLib PRIVATE third_party/PawnIO/PawnIOLib/include third_party/PawnIO/PawnIO/include)
target_compile_definitions(PawnIOLib PRIVATE PawnIOLib_EXPORTS PAWNIO_NAME="PawnIO")
target_link_libraries(PawnIOLib PRIVATE ntdll)
add_library(mini_core STATIC src/core.cpp src/config.cpp src/sensors.cpp src/device.cpp src/engine.cpp src/autostart.cpp)
target_include_directories(mini_core PUBLIC src third_party)
target_compile_definitions(mini_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(mini_core PRIVATE /W4 /utf-8 /EHsc /permissive- /Gy /Gw)
target_link_libraries(mini_core PUBLIC leveldb snappy setupapi hid pdh psapi shell32 ole32 comctl32 user32 advapi32 oleaut32 taskschd)
add_executable(MythCool-mini WIN32 src/main.cpp src/app.manifest src/resources.rc)
target_include_directories(MythCool-mini PRIVATE assets)
target_link_libraries(MythCool-mini PRIVATE mini_core)
add_dependencies(MythCool-mini PawnIOLib)
target_compile_options(MythCool-mini PRIVATE /W4 /utf-8 /EHsc)
target_link_options(MythCool-mini PRIVATE /OPT:REF /OPT:ICF /MANIFEST:EMBED "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'" "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/src/app.manifest")
add_custom_command(TARGET MythCool-mini POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/AMDFamily17.bin" "$<TARGET_FILE_DIR:MythCool-mini>/AMDFamily17.bin")
enable_testing()
add_executable(mini_tests tests/core_tests.cpp)
target_link_libraries(mini_tests PRIVATE mini_core)
target_compile_options(mini_tests PRIVATE /utf-8 /EHsc)
add_test(NAME mini_core COMMAND mini_tests)

add_executable(sensor_bench EXCLUDE_FROM_ALL tools/sensor_bench.cpp)
target_link_libraries(sensor_bench PRIVATE mini_core)
target_compile_options(sensor_bench PRIVATE /utf-8 /EHsc)
